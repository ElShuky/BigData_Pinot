-- EXPERIMENTO 1 (Tabla única progresiva: security_events_exp1)
-- Copia y pega estas 3 consultas en http://localhost:9000/#/query
-- en cada etapa: Config A -> Config B (tras Reload) -> Config C (tras Reload)
-- =============================================================================

-- Q1 (Bloom Filter - Poda de segmentos en src_ip):
-- La IP 172.20.45.99 solo existe en 1 de los 20 segmentos (part_10.json).
-- Observar: timeUsedMs, numSegmentsPrunedByValue, numSegmentsProcessed, numEntriesScannedInFilter
SELECT count(*) AS total_eventos, sum(bytes_transferred) AS total_bytes
FROM security_events_exp1
WHERE src_ip = '172.20.45.99';

-- Q2 (Inverted Index - Filtrado exacto con Roaring Bitmaps):
-- Filtra por event_type y severity (indexadas en Config B y C).
-- Observar: timeUsedMs, numEntriesScannedInFilter (millones en A -> 0 en B y C)
SELECT username, count(*) AS total_intentos
FROM security_events_exp1
WHERE event_type = 'login_fail' AND severity = 'HIGH'
GROUP BY username
ORDER BY total_intentos DESC;

-- Q3 (Star-Tree Index - Pre-agregación multidimensional OLAP):
-- Agrega count, sum y avg pre-calculados en el árbol Star-Tree de Config C.
-- Observar: timeUsedMs, numDocsScanned (millones en A y B -> ~120 en C)
SELECT
    event_type,
    severity,
    count(*) AS total_eventos,
    sum(bytes_transferred) AS total_bytes,
    avg(bytes_transferred) AS promedio_bytes
FROM security_events_exp1
GROUP BY event_type, severity
ORDER BY total_eventos DESC;
